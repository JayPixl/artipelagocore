Config: {
  settings: { } // Nothing yet, will add later
  presets: Preset[]
  schedules: Schedule[]
}

Preset: {
  id: String
  name: String
  description: String[]
  effects: Effect[]
}

Schedule: {
  type: "weekly"
  day: "sunday" | "monday" | "tuesday" | "wednesday" | "thursday" | "friday" | "saturday"
  start: TimeUTC
  end: TimeUTC
  preset: String    // References Preset ID
} | {
  type: "once"
  start: DateTimeUTC
  end: DateTimeUTC
  preset: String
}

Effect: {
  type: "shiny_boost"
  targets: Pokemon[]
  contexts: SpawnContext[]
  operation: "set" | "multiply"
  value: Double
} | {
  type: "hidden_ability"
  targets: Pokemon[]
  contexts: SpawnContext[]
  chance: Double  // Chance is always 0 <= p <= 1
} | {
  type: "knows_move"
  targets: Pokemon[]
  contexts: SpawnContext[]
  move: Move     // Move's String ID
  chance: Double
} | {
  type: "min_ivs"
  targets: Pokemon[]
  contexts: SpawnContext[]
  value: Int
} | {
  type: "add_spawns"
  spawns: SpawnEntry[]
} | {
  type: "multiply_spawn_weight"
  targets: Pokemon[]
  multiplier: Double
} | {
  type: "modify_global_bucket_weights"
  buckets: BucketAssignment[]
} | {
  type: "player_xp_boost"
  contexts: PlayerXPGainContext[]
  multiplier: Double
} | {
  type: "pokemon_xp_boost"
  contexts: PokemonXPGainContext[]
  multiplier: Double
} | {
  type: "pokemon_ev_boost"
  contexts: PokemonEVGainContext[]
  multiplier: Double
  chance: Double
} | {
  type: "loot_inject"
  contexts: LootInjectContext[]
  items: Item[]
}

Pokemon: {
  species: Species | "all",   // Species is String
  aspects: Aspect[]   // Aspect is String
}

SpawnContext: {
  biome: BiomeTag | Biome | "all",        // Defaults to "all"
  cause: "all" | "fishing" | "normal" | "bait"
}

SpawnEntry: {
  id: String                 // Unique ID for this event spawn entry
  pokemon: Species            // Cobblemon species ID
  context: "grounded" | "submerged" | "surface" | "fishing"
  bucket: "common" | "uncommon" | "rare" | "ultra-rare"
  weight: Double
  level: String               // Cobblemon level range, e.g. "30-40"
  condition?: SpawnCondition  // Optional native-style spawn restrictions
  anticondition?: SpawnCondition
  aspects?: Aspect[]
  forms?: String[]
  heldItems?: HeldItem[]
}

SpawnCondition: {
  biomes?: (BiomeTag | Biome)[]
  timeRange?: "day" | "night"
  moonPhase?: String
  minY?: Int
  maxY?: Int
  canSeeSky?: Boolean
}

HeldItem: {
  item: Item
  percentage: Double
}

BucketAssignment: {
  bucket: "common" | "uncommon" | "rare" | "ultra-rare"
  weight: Double
}

PlayerXPGainContext: { // More will be added later, just these for now
  type: "all"
} | {
  type: "fishing"
} | {
  type: "break_block"
  blocks: String[]    // IDs of blocks
}

PokemonXPGainContext: {  // More can be added later, just these for now
  type: "all"
} | {
  type: "wild_battle"
  pokemon: Pokemon[]    // The defeated pokemon
}

PokemonEVGainContext: {   // More can be added later, just these for now
  type: "all"
} | {
  type: "wild_battle"
  pokemon: Pokemon[]    // The defeated pokemon
} | {
  type: "vitamin"
}

LootInjectContext: {     // More will be added later
  type: "defeat_pokemon"
  pokemon: Pokemon[]     // The defeated pokemon
  defeat_context: "fishing" | "normal" | "bait"
} | {
  type: "catch_pokemon"
  pokemon: Pokemon[]
  catch_context: "fishing" | "normal" | "bait"
}

Item: {
  id: String
  amount: [min, max]
  chance: Double
}



==== Commands ====
/events
  list - Lists active events with names and descriptions
    detail - admin only, lists events with ids, names, and descriptions
  start <preset> <minutes> - admin only, starts chosen preset for chosen duration
  stop <preset> - admin only, immediately stops chosen preset
  shutdown - admin only, halts all schedules and active effects
  reload - admin only, reloads and refreshes the schedules
